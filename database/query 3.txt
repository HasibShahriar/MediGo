USE MediGoDB;

CREATE TABLE ContactMessages
(
    Id INT IDENTITY(1,1) PRIMARY KEY,

    PatientId INT NOT NULL,

    Name NVARCHAR(150) NOT NULL,

    Email NVARCHAR(200) NOT NULL,

    Concern NVARCHAR(100) NOT NULL,

    Message NVARCHAR(1000) NOT NULL,

    IsRead BIT NOT NULL DEFAULT 0,

    CreatedAt DATETIME2 NOT NULL DEFAULT GETDATE(),

    CONSTRAINT FK_ContactMessages_Patients
        FOREIGN KEY (PatientId)
        REFERENCES Patients(Id)
);


CREATE TABLE Medicines
(
    Id INT IDENTITY(1,1) PRIMARY KEY,

    Name NVARCHAR(150) NOT NULL,

    GenericName NVARCHAR(150) NOT NULL,

    BrandName NVARCHAR(150) NULL,

    Manufacturer NVARCHAR(200) NOT NULL,

    Category NVARCHAR(100) NOT NULL,

    Strength NVARCHAR(50) NOT NULL,

    DosageForm NVARCHAR(50) NOT NULL,

    PackSize NVARCHAR(100) NULL,

    Price DECIMAL(10,2) NOT NULL,

    StockQuantity INT NOT NULL
        CONSTRAINT DF_Medicines_Stock DEFAULT 0,

    PrescriptionRequired BIT NOT NULL
        CONSTRAINT DF_Medicines_Prescription DEFAULT 0,

    Description NVARCHAR(1000) NULL,

    Image NVARCHAR(500) NULL,

    IsAvailable BIT NOT NULL
        CONSTRAINT DF_Medicines_Available DEFAULT 1,

    CreatedAt DATETIME2 NOT NULL
        CONSTRAINT DF_Medicines_CreatedAt DEFAULT GETDATE(),

    CONSTRAINT CK_Medicines_Price
        CHECK (Price >= 0),

    CONSTRAINT CK_Medicines_Stock
        CHECK (StockQuantity >= 0)
);


CREATE TABLE CartItems
(
    Id INT IDENTITY(1,1) PRIMARY KEY,

    PatientId INT NOT NULL,

    MedicineId INT NOT NULL,

    Quantity INT NOT NULL DEFAULT 1,

    CreatedAt DATETIME2 NOT NULL DEFAULT GETDATE(),

    UpdatedAt DATETIME2 NOT NULL DEFAULT GETDATE(),

    CONSTRAINT FK_CartItems_Patients
        FOREIGN KEY (PatientId)
        REFERENCES Patients(Id)
        ON DELETE CASCADE,

    CONSTRAINT FK_CartItems_Medicines
        FOREIGN KEY (MedicineId)
        REFERENCES Medicines(Id)
        ON DELETE CASCADE,

    CONSTRAINT UQ_CartItems_Patient_Medicine
        UNIQUE (PatientId, MedicineId),

    CONSTRAINT CK_CartItems_Quantity
        CHECK (Quantity > 0)
);

select * from ContactMessages;
select * from Medicines;
select * from CartItems;

SELECT
    c.Id AS CartItemId,
    c.PatientId,
    c.MedicineId,
    c.Quantity,

    m.Name,
    m.GenericName,
    m.Strength,
    m.Price

FROM CartItems c

INNER JOIN Medicines m
    ON c.MedicineId = m.Id

WHERE c.PatientId = 5;


IF COL_LENGTH('Patients', 'CreatedAt') IS NULL
BEGIN
    ALTER TABLE Patients
    ADD CreatedAt DATETIME2 NOT NULL
        CONSTRAINT DF_Patients_CreatedAt
        DEFAULT GETDATE();
END
